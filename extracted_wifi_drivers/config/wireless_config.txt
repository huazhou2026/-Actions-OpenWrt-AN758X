# /etc/config/wireless - WiFi Configuration
# Device: Nokia XG-040G-MD (UBI)
# Firmware: ImmortalWrt

config wifi-device 'radio0'
	option type 'mac80211'
	option path 'platform/soc/1fad0000.usb/usb3/3-1/3-1:1.3'
	option country 'CN'
	option cell_density '0'
	option disabled '1'

config wifi-device 'radio1'
	option type 'mac80211'
	option path 'platform/soc/1fad0000.usb/usb3/3-1/3-1:1.0'
	option band '5g'
	option cell_density '0'
	option country 'CN'
	option channel '153'

config wifi-iface 'wifinet0'
	option device 'radio1'
	option mode 'ap'
	option ssid 'ImmortalWrt'
	option encryption 'none'
	option network 'lan'
